```python
"""

import base64
import binascii
import os

from cryptography.exceptions import InvalidTag
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
from cryptography.hazmat.primitives.kdf.scrypt import Scrypt


_VERSION = 1

_SALT_SIZE = 16
_NONCE_SIZE = 12
_KEY_SIZE = 32  # AES-256


def _derive_key(password: str, salt: bytes) -> bytes:
    """Derive a 256-bit AES key from a password/string key."""
    kdf = Scrypt(
        salt=salt,
        length=_KEY_SIZE,
        n=2**14,
        r=8,
        p=1,
    )
    return kdf.derive(password.encode("utf-8"))


def encrypt(key: str, clear: str) -> str:
    if not isinstance(key, str):
        raise TypeError("key must be a string")

    if not key:
        raise ValueError("key must not be empty")

    if not isinstance(clear, str):
        raise TypeError("clear must be a string")

    salt = os.urandom(_SALT_SIZE)
    nonce = os.urandom(_NONCE_SIZE)

    aes_key = _derive_key(key, salt)

    ciphertext = AESGCM(aes_key).encrypt(
        nonce,
        clear.encode("utf-8"),
        None,
    )

    payload = (
        bytes([_VERSION])
        + salt
        + nonce
        + ciphertext
    )

    return base64.urlsafe_b64encode(payload).decode("ascii")


def decrypt(key: str, enc: str) -> str:
    if not isinstance(key, str):
        raise TypeError("key must be a string")

    if not key:
        raise ValueError("key must not be empty")

    if not isinstance(enc, str):
        raise TypeError("enc must be a string")

    if not enc:
        raise ValueError("enc must not be empty")

    try:
        payload = base64.b64decode(
            enc.encode("ascii"),
            altchars=b"-_",
            validate=True,
        )
    except (UnicodeEncodeError, binascii.Error, ValueError) as exc:
        raise ValueError("invalid encrypted value") from exc

    # 1-byte version
    # 16-byte salt
    # 12-byte nonce
    # >=16-byte GCM authentication tag
    minimum_size = 1 + _SALT_SIZE + _NONCE_SIZE + 16

    if len(payload) < minimum_size:
        raise ValueError("invalid encrypted value")

    version = payload[0]

    if version != _VERSION:
        raise ValueError(
            f"unsupported encrypted value version: {version}"
        )

    salt_start = 1
    nonce_start = salt_start + _SALT_SIZE
    ciphertext_start = nonce_start + _NONCE_SIZE

    salt = payload[salt_start:nonce_start]
    nonce = payload[nonce_start:ciphertext_start]
    ciphertext = payload[ciphertext_start:]

    aes_key = _derive_key(key, salt)

    try:
        clear = AESGCM(aes_key).decrypt(
            nonce,
            ciphertext,
            None,
        )

        return clear.decode("utf-8")

    except (InvalidTag, UnicodeDecodeError) as exc:
        raise ValueError(
            "decryption failed: wrong key or corrupted data"
        ) from exc

key = "my-secret-key" #petx2

encrypted = encrypt(
    key,
    "Uzumaki Naruto",
)

print(encrypted)

decrypted = decrypt(key, encrypted)

print(decrypted)
```

```lua
    -- Convert HSL to RGB
    local function hslToRgb(h, s, l)
        local function f(n)
            local k = (n + h * 12) % 12
            local a = s * math.min(l, 1 - l)
            return l - a * math.max(-1, math.min(math.min(k - 3, 9 - k), 1))
        end
        return { red = f(0), green = f(8), blue = f(4) }
    end
    
    -- Generate color: highlight if active
    local function generateColor(index, isActive)

                return true
            end
        end)

                    key = tostring(r.index)

                    key = tostring(r.index - 5)
                    mods = { "alt" }
                end
                -- alert.show(table.concat(mods, "+") .. "+" .. key)
                eventtap.keyStroke(mods, key)
                return false
            end
        end
        return false
    end)
```
